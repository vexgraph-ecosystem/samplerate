cmake_minimum_required(VERSION 3.24)
# Standalone CLion metadata only. Actual builds belong to b; no fetch or link.
# Missing headers stay real errors; no fake declarations are generated.
if(APPLE)
    set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "Supported architecture")
    set(CMAKE_OSX_DEPLOYMENT_TARGET 14.0 CACHE STRING "Supported OS floor")
endif()
project(samplerate_ide LANGUAGES C OBJC)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED YES)
set(CMAKE_C_EXTENSIONS YES)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(VEXSPOKE_SOURCE_DIR "" CACHE PATH "Local Vexspoke src; never downloaded")
set(RELATIONAL_ENGINE_SOURCE_DIR "" CACHE PATH "Local Relational Engine src for canonical nio/io/type headers; never downloaded")
file(GLOB_RECURSE sources CONFIGURE_DEPENDS "src/*.c" "src/*.m" "src/*.h")
list(FILTER sources EXCLUDE REGEX "/(attic|vendor|third_party|_old)/")
# Only one platform stream backend compiles per host; the IDE model shows the
# one producing a target on this OS.
if(APPLE)
    list(FILTER sources EXCLUDE REGEX "audio_stream_stub\\.c$")
else()
    list(FILTER sources EXCLUDE REGEX "audio_stream_cocoa\\.m$")
endif()
add_library(samplerate_ide OBJECT EXCLUDE_FROM_ALL ${sources})
target_include_directories(samplerate_ide PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
foreach(path VEXSPOKE_SOURCE_DIR RELATIONAL_ENGINE_SOURCE_DIR)
    if(${path})
        target_include_directories(samplerate_ide PRIVATE "${${path}}")
    endif()
endforeach()
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(samplerate_ide PRIVATE -std=gnu23 -Wall -Wextra -Werror)
    if(APPLE)
        target_compile_options(samplerate_ide PRIVATE -mcpu=apple-m1)
    endif()
endif()
target_compile_options(samplerate_ide PRIVATE $<$<COMPILE_LANGUAGE:OBJC>:-fobjc-arc>)
message(STATUS "IDE metadata only: excluded object target. Use b for builds; missing headers stay errors.")
